cmake_minimum_required(VERSION 3.16)
project(dutyon_backend_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(JSON_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../third_party/deps/json-3.11.3/include"
    CACHE PATH "Directory containing nlohmann/json.hpp")
add_executable(backend_tests backend_tests.cpp ../src/backend/state_manager.cpp)
target_include_directories(backend_tests PRIVATE ../src "${JSON_INCLUDE_DIR}")
if(MSVC)
    target_compile_options(backend_tests PRIVATE /utf-8)
endif()
if(WIN32)
    target_sources(backend_tests PRIVATE ../src/backend/hooks_installer.cpp ../src/backend/ide_scanner.cpp)
    target_compile_definitions(backend_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(backend_tests PRIVATE user32)
endif()
add_test(NAME backend_tests COMMAND backend_tests "${CMAKE_CURRENT_LIST_DIR}/../../hooks")
